cmake_minimum_required(VERSION 3.13)
project(crashbox C)

# core: portable C99, no heap, no locks, no OS calls
add_library(crashbox STATIC src/crashbox.c)
target_include_directories(crashbox PUBLIC include)
set_property(TARGET crashbox PROPERTY C_STANDARD 99)

# optional: the usual policy (boot report, checkpoints, deferred writes, routes)
add_library(crashbox_service STATIC src/crashbox_service.c)
target_link_libraries(crashbox_service PUBLIC crashbox)

# optional: storage in one fixed-size file (POSIX)
add_library(crashbox_file STATIC src/crashbox_file.c)
target_link_libraries(crashbox_file PUBLIC crashbox)
target_compile_definitions(crashbox_file PRIVATE _DEFAULT_SOURCE)

# optional: tiny HTTP readout (BSD sockets; define CBX_HTTP_LWIP for lwIP)
add_library(crashbox_http STATIC src/crashbox_http.c)
target_link_libraries(crashbox_http PUBLIC crashbox)
target_compile_definitions(crashbox_http PRIVATE _DEFAULT_SOURCE)

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    option(CRASHBOX_TESTS "Build the host tests" ON)
else()
    option(CRASHBOX_TESTS "Build the host tests" OFF)
endif()
if(CRASHBOX_TESTS)
    enable_testing()
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        foreach(t crashbox crashbox_service crashbox_file crashbox_http)
            target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
        endforeach()
    endif()
    add_executable(test_crashbox tests/test_crashbox.c)
    target_link_libraries(test_crashbox PRIVATE crashbox_file)
    add_executable(test_service tests/test_service.c)
    target_link_libraries(test_service PRIVATE crashbox_service crashbox_http)
    target_compile_definitions(test_service PRIVATE _DEFAULT_SOURCE)
    add_test(NAME crashbox COMMAND test_crashbox WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
    add_test(NAME crashbox_service COMMAND test_service)
endif()
